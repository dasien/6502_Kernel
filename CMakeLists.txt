cmake_minimum_required(VERSION 3.20)
project(6502-kernel 
    VERSION 1.0.0
    DESCRIPTION "6502 Microprocessor Emulator with Interactive Monitor"
    LANGUAGES CXX
)

# Set C++ standard
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Organize build outputs
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

# Build configuration
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

message(STATUS "6502 Computer Emulator v${PROJECT_VERSION}")
message(STATUS "Build type: ${CMAKE_BUILD_TYPE}")
message(STATUS "========================================")

# Compiler-specific options
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wall -Wextra -Wpedantic)
    if(CMAKE_BUILD_TYPE STREQUAL "Debug")
        add_compile_options(-g -O0 -DDEBUG)
    else()
        add_compile_options(-O3 -DNDEBUG)
    endif()
endif()

# Include CMake modules
list(APPEND CMAKE_MODULE_PATH ${CMAKE_SOURCE_DIR}/tools/cmake)

# Find Qt (optional GUI support)
include(FindQt)

# Add subdirectories
add_subdirectory(src)

# Kernel ROM build (if cc65 available)
include(BuildKernel)

# Testing support
option(BUILD_TESTS "Build unit tests" OFF)
if(BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

# ----------------------------------------------------------------
# mkfat16 - host tool to create a FAT16 disk.img for the block device.
# Reuses the FAT16 image builder shared with the filesystem tests.
# Build it, then create a disk:   ./bin/mkfat16 ../disk.img [files...]
# Or use the convenience target:  ninja sample_disk
# ----------------------------------------------------------------
add_executable(mkfat16 tools/mkfat16/mkfat16.cpp)
target_include_directories(mkfat16 PRIVATE ${CMAKE_SOURCE_DIR}/tests/support)
target_compile_features(mkfat16 PRIVATE cxx_std_20)

# ----------------------------------------------------------------
# mkdisk - diskmap-driven disk image tool (create / read / update). Reuses the
# FAT16 builder + reader shared with the tests. Supersedes mkfat16.
#   ./bin/mkdisk create <image> <diskmap.txt>
#   ./bin/mkdisk read   <image> <outdir>
#   ./bin/mkdisk update <image> <diskmap.txt>
# ----------------------------------------------------------------
add_executable(mkdisk tools/mkdisk/mkdisk.cpp)
target_include_directories(mkdisk PRIVATE ${CMAKE_SOURCE_DIR}/tests/support)
target_compile_features(mkdisk PRIVATE cxx_std_20)

# ----------------------------------------------------------------
# mkprg - prepend the 2-byte little-endian load-address header to a linked
# binary, making it a .PRG the DOS can launch.
#   ./bin/mkprg 0800 venture.bin VENTURE.PRG
# ----------------------------------------------------------------
add_executable(mkprg tools/mkprg/mkprg.cpp)
target_compile_features(mkprg PRIVATE cxx_std_20)

# Assemble the default disk.img the GUI loads (${CMAKE_BINARY_DIR}/disk.img).
# Explicit target -- NOT part of `ninja`/ALL -- so routine builds never rewrite the
# disk. The programs are ordinary targets, so `ninja disk` rebuilds whichever of
# them you changed and no more.
#
# Both the staging commands and the diskmap mkdisk reads are derived from
# programs/catalog.txt; neither is written by hand any more.
include(${CMAKE_SOURCE_DIR}/tools/cmake/DiskCatalog.cmake)
mfc_read_catalog(${CMAKE_SOURCE_DIR}/programs/catalog.txt)
mfc_check_catalog_covers_builds()

set(DISKROOT ${CMAKE_BINARY_DIR}/diskroot)
set(GENERATED_DISKMAP ${CMAKE_BINARY_DIR}/diskmap.generated.txt)

set(DISK_STAGE_COMMANDS "")
set(DISK_DRAWERS "")
# Built as one string, not a list: a ';' anywhere in it would be read as a CMake list
# separator and silently vanish from the generated file.
set(DISK_MANIFEST "# GENERATED by cmake from programs/catalog.txt -- do not edit.\n")
string(APPEND DISK_MANIFEST
       "# One disk path per line. mkdisk reads each file from <diskmap-dir>/<path>.\n")

foreach(entry IN LISTS MFC_CAT_NAMES)
    foreach(spec IN LISTS MFC_CAT_${entry}_FILES)
        string(REPLACE "|" ";" parts "${spec}")
        list(GET parts 0 src)
        list(GET parts 1 dst)
        list(GET parts 2 kind)
        # A built entry's `program` IS the build output, so it comes from the build
        # tree; its `data`/`doc` are committed content and still come from the source
        # tree. Entries with no build script (the Scott Adams games) are committed
        # whole and unaffected.
        if(MFC_CAT_${entry}_CONFIG AND kind STREQUAL "program")
            set(srcroot ${CMAKE_BINARY_DIR}/${MFC_CAT_${entry}_DIR})
        else()
            set(srcroot ${CMAKE_SOURCE_DIR}/${MFC_CAT_${entry}_DIR})
        endif()
        get_filename_component(drawer "${dst}" DIRECTORY)
        if(drawer)
            list(APPEND DISK_DRAWERS "${drawer}")
        endif()
        list(APPEND DISK_STAGE_COMMANDS
             COMMAND ${CMAKE_COMMAND} -E copy ${srcroot}/${src} ${DISKROOT}/${dst})
        string(APPEND DISK_MANIFEST "${dst}\n")
    endforeach()
endforeach()

if(DISK_DRAWERS)
    list(REMOVE_DUPLICATES DISK_DRAWERS)
endif()
set(DISK_MKDIR_COMMANDS "")
foreach(drawer IN LISTS DISK_DRAWERS)
    list(APPEND DISK_MKDIR_COMMANDS
         COMMAND ${CMAKE_COMMAND} -E make_directory ${DISKROOT}/${drawer})
endforeach()

file(WRITE ${GENERATED_DISKMAP} ${DISK_MANIFEST})

# ----------------------------------------------------------------------------
# The .PRG files themselves.
#
# These are build outputs, not committed artifacts, so nothing can stage a stale one
# -- the failure this used to have was a program's C being edited, the .PRG not being
# rebuilt, and the disk carrying yesterday's binary while the test blobs (built from
# the same sources by CMake) picked the change up. The tests agreed with you and the
# machine did not. With the .PRGs built here that state cannot exist.
#
# One target per catalog entry with a build recipe, each compiling its sources to
# objects in the build tree and linking them; see tools/cmake/Programs.cmake.
set(PROGRAM_TARGETS "")
if(CL65_FOUND)
    include(${CMAKE_SOURCE_DIR}/tools/cmake/Programs.cmake)
    mfc_add_catalog_programs(PROGRAM_TARGETS)
else()
    # cc65 is required to build at all (the ROMs are ca65 assembly and none are
    # committed), so this is a broken-install path rather than a supported one. Say so
    # plainly instead of failing in the staging copy with a missing-file error.
    add_custom_target(disk_programs_missing
        COMMAND ${CMAKE_COMMAND} -E echo
                "cc65 not found - cannot build the disk programs"
    )
    set(PROGRAM_TARGETS disk_programs_missing)
endif()

# `ninja programs` - every .PRG, without assembling a disk.
add_custom_target(programs COMMENT "Building the disk programs")
add_dependencies(programs ${PROGRAM_TARGETS})

# `ninja disk` - build the programs, then assemble disk.img from the catalog.
add_custom_target(disk
    COMMAND ${CMAKE_COMMAND} -E rm -rf ${DISKROOT}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${DISKROOT}
    ${DISK_MKDIR_COMMANDS}
    ${DISK_STAGE_COMMANDS}
    COMMAND ${CMAKE_COMMAND} -E copy ${GENERATED_DISKMAP} ${DISKROOT}/diskmap.txt
    COMMAND mkdisk create ${CMAKE_BINARY_DIR}/disk.img ${DISKROOT}/diskmap.txt
    DEPENDS mkdisk
    COMMENT "Assembling disk.img"
    USES_TERMINAL
    VERBATIM
)
# Target-level, so ninja builds every .PRG before the staging copies run.
add_dependencies(disk ${PROGRAM_TARGETS})

# `ninja everything` - the whole machine: the disk, plus the ROMs and the app. All of
# those are ALL targets, so a plain `ninja` builds them; naming them here is what
# makes `ninja everything` on its own sufficient.
add_custom_target(everything
    COMMENT "Building every program, ROM and the disk"
)
set(EVERYTHING_DEPS disk mkdisk 6502-kernel)
foreach(t kernel_rom dos_rom basic_rom forth_rom monitor_rom
          xmodem_bin term_bin irc_bin edit_bin venture_bin)
    if(TARGET ${t})
        list(APPEND EVERYTHING_DEPS ${t})
    endif()
endforeach()
add_dependencies(everything ${EVERYTHING_DEPS})

# `ninja run` - build everything, then boot it.
#
# From bin/, because two paths the machine uses are relative to the working
# directory: ../kernel/kernel.rom and ../disk.img.
add_custom_target(run
    COMMAND ${CMAKE_COMMAND} -E chdir ${CMAKE_BINARY_DIR}/bin ./6502-kernel
    DEPENDS everything
    COMMENT "Launching the MFC 6502"
    USES_TERMINAL
    VERBATIM
)

# Installation rules
install(TARGETS 6502-kernel
    RUNTIME DESTINATION bin
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
)

install(FILES ${CMAKE_BINARY_DIR}/kernel/kernel.rom
    DESTINATION share/6502-kernel
    OPTIONAL
)

# Desktop integration (Linux). The icon goes into the hicolor theme at each size
# the generator produces, which is what desktops and task switchers look up by the
# .desktop file's "Icon=mfc6502" name. Harmless elsewhere: the files simply are not
# consulted. The icon is also compiled into the binary (include/ui/AppIcon.h), so
# the running window has one whether or not these are installed.
if(UNIX AND NOT APPLE)
    foreach(icon_size 16 32 48 64 128 256 512)
        install(FILES ${CMAKE_SOURCE_DIR}/assets/mfc6502-${icon_size}.png
            DESTINATION share/icons/hicolor/${icon_size}x${icon_size}/apps
            RENAME mfc6502.png
            OPTIONAL
        )
    endforeach()
    install(FILES ${CMAKE_SOURCE_DIR}/assets/mfc6502.desktop
        DESTINATION share/applications
        OPTIONAL
    )
endif()
